  if(lPunktow<60){

double srednia(List<int> liczby){
  double suma=0;
  for(var el in liczby){
    suma+=el;
  }
  
  if(liczby.length>0){
    return suma/liczby.length;
  }
  else return 0.0;
}
void main() {

final liczby=<int>[1,2,3,4];
  print(srednia(liczby));
final liczby2=<int>[1,2,2,2,4];
  print(srednia(liczby2));
final liczby3=<int>[1,2,3,4,1,1,5];
  print(srednia(liczby3));
      
  
}

class Student {
  String? imie;
  int? rok;
  double? srednia;
  
  void opis(){
    print("student $imie, rok $rok, srednia $srednia");
  }
  Student([this.imie, this.rok, this.srednia]);
   
} 

void main() {

  Student s1=Student('Janusz',4,3.5);
  Student s2=Student('Marysia',4,3.5);
  s1.opis();
  s2.opis();
  
}
